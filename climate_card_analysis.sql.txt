-- ============================================================
-- 기후동행카드 정책 효과 분석 SQL
-- 분석 도구: DBeaver (SQLite)
-- 테이블 목록:
--   subway_usage      : 지하철 일별 승하차 데이터 (2022~2026)
--   climate_card      : 기후동행카드 월별 활성화 카드 수
--   transport_cost    : 시도별 월평균 대중교통 비용 구간 비율
--   population        : 시도별 인구 현황
-- transport_cost 컬럼 안내:
--   under2=2만원미만, w2to4=2~4만원, w4to6=4~6만원,
--   w6to8=6~8만원, w8to10=8~10만원, w10to12=10~12만원, over12=12만원이상
-- ============================================================


-- ============================================================
-- [사전확인] 전체 데이터 기간 확인
-- ============================================================
SELECT
    MIN(사용일자) AS 시작일,
    MAX(사용일자) AS 종료일,
    COUNT(DISTINCT 사용일자) AS 총일수
FROM subway_usage;


-- ============================================================
-- 분석 1. 서울 지하철 월별 이용량 추이 (2022~2025)
-- 목적: 기후동행카드 시행 전후 서울 지하철 이용량 변화 파악
-- ============================================================
SELECT
    STRFTIME('%Y-%m', 사용일자)         AS 기준연월,
    SUM(승차총승객수)                    AS 월별_승차합계,
    SUM(하차총승객수)                    AS 월별_하차합계,
    SUM(승차총승객수 + 하차총승객수)     AS 월별_총이용객수
FROM subway_usage
WHERE 사용일자 >= '2022-01-01'
  AND 사용일자 <= '2025-12-31'
GROUP BY 기준연월
ORDER BY 기준연월;


-- ============================================================
-- 분석 2. 기후동행카드 시행 전후 연간 이용객 비교
-- 목적: 정책 전(2022~2023) vs 정책 후(2024~2025) 이용량 비교
-- ============================================================
SELECT
    STRFTIME('%Y', 사용일자) AS 연도,
    CASE
        WHEN STRFTIME('%Y', 사용일자) IN ('2022','2023') THEN '정책시행전'
        WHEN STRFTIME('%Y', 사용일자) IN ('2024','2025') THEN '정책시행후'
    END AS 정책구분,
    SUM(승차총승객수 + 하차총승객수)            AS 연간_총이용객수,
    ROUND(AVG(승차총승객수 + 하차총승객수), 0) AS 일평균_이용객수
FROM subway_usage
WHERE 사용일자 BETWEEN '2022-01-01' AND '2025-12-31'
GROUP BY 연도
ORDER BY 연도;


-- ============================================================
-- 분석 3. 기후동행카드 활성카드 수 vs 월별 이용량 비교
-- 목적: 카드 보급 증가와 지하철 이용량 증가의 연관성 확인
-- ============================================================

SELECT
    c.yyyy || '-' || PRINTF('%02d', c.mm) AS 기준연월,
    c.active_cards,
    ROUND(CAST(c.active_cards AS REAL) / 10000, 1) AS 활성카드수_만장,
    SUM(s.승차총승객수 + s.하차총승객수) AS 월별_총이용객수
FROM climate_card c
JOIN subway_usage s
  ON c.yyyy || '-' || PRINTF('%02d', c.mm) = SUBSTR(s.사용일자, 1, 7)
GROUP BY c.yyyy, c.mm, c.active_cards
ORDER BY c.yyyy, c.mm;
-- ============================================================
-- 분석 4. 같은 월 기준 2023 vs 2024 이용량 비교 (YoY)
-- 목적: 계절 효과 제거 후 순수 정책 효과 측정
-- ============================================================
SELECT
    a.기준월,
    a.이용객수_2023 AS 시행전_2023,
    b.이용객수_2024 AS 시행후_2024,
    ROUND((b.이용객수_2024 - a.이용객수_2023) * 100.0
          / a.이용객수_2023, 2)          AS 증감률_퍼센트
FROM (
    SELECT
        STRFTIME('%m', 사용일자) AS 기준월,
        SUM(승차총승객수 + 하차총승객수) AS 이용객수_2023
    FROM subway_usage
    WHERE 사용일자 BETWEEN '2023-01-01' AND '2023-12-31'
    GROUP BY 기준월
) a
JOIN (
    SELECT
        STRFTIME('%m', 사용일자) AS 기준월,
        SUM(승차총승객수 + 하차총승객수) AS 이용객수_2024
    FROM subway_usage
    WHERE 사용일자 BETWEEN '2024-01-01' AND '2024-12-31'
    GROUP BY 기준월
) b ON a.기준월 = b.기준월
ORDER BY a.기준월;


-- ============================================================
-- 분석 5. 노선별 이용량 변화 (정책 전후 비교)
-- 목적: 어떤 노선에서 이용량 증가가 두드러졌는지 파악
-- ============================================================
SELECT
    노선명,
    SUM(CASE WHEN 사용일자 BETWEEN '2023-02-01' AND '2023-12-31'
             THEN 승차총승객수 + 하차총승객수 ELSE 0 END) AS 시행전_이용객수,
    SUM(CASE WHEN 사용일자 BETWEEN '2024-02-01' AND '2024-12-31'
             THEN 승차총승객수 + 하차총승객수 ELSE 0 END) AS 시행후_이용객수,
    ROUND(
        (SUM(CASE WHEN 사용일자 BETWEEN '2024-02-01' AND '2024-12-31'
                  THEN 승차총승객수 + 하차총승객수 ELSE 0 END)
       - SUM(CASE WHEN 사용일자 BETWEEN '2023-02-01' AND '2023-12-31'
                  THEN 승차총승객수 + 하차총승객수 ELSE 0 END))
        * 100.0
        / NULLIF(SUM(CASE WHEN 사용일자 BETWEEN '2023-02-01' AND '2023-12-31'
                          THEN 승차총승객수 + 하차총승객수 ELSE 0 END), 0)
    , 2) AS 증감률_퍼센트
FROM subway_usage
GROUP BY 노선명
HAVING 시행전_이용객수 > 0 AND 시행후_이용객수 > 0
ORDER BY 증감률_퍼센트 DESC;


-- ============================================================
-- 분석 6. 시도별 교통비 부담 구조 비교
-- 목적: 서울과 타 지역의 교통비 부담 차이 파악
-- ============================================================

SELECT 지역,
    under2 AS 저비용_2만미만_PCT,
    ROUND(w2to4 + w4to6, 1) AS 중비용_2_6만_PCT,
    ROUND(w6to8 + w8to10, 1) AS 중고비용_6_10만_PCT,
    ROUND(w10to12 + over12, 1) AS 고비용_10만이상_PCT,
    ROUND(w6to8 + w8to10 + w10to12 + over12, 1) AS 월6만이상_고부담_PCT
FROM transport_cost
ORDER BY 월6만이상_고부담_PCT DESC;
-- ============================================================
-- 분석 7. 서울 vs 타지역 교통비 고부담 비율 비교
-- 목적: 타 지역 기후동행카드 필요성 정량 비교
-- ============================================================
SELECT
    지역,
    ROUND(w6to8 + w8to10 + w10to12 + over12, 1) AS 월6만이상_PCT,
    CASE
        WHEN 지역 = '서울' THEN '기후동행카드_시행중'
        WHEN 지역 = '전체' THEN '전국평균'
        ELSE '미시행지역'
    END AS 정책현황
FROM transport_cost
ORDER BY 월6만이상_PCT DESC;


-- ============================================================
-- 분석 8. 인구 규모 + 교통비 고부담 비율 결합 분석
-- 목적: 기후동행카드 정책 확대 우선순위 지역 도출
-- ============================================================
SELECT
    p.시도,
    p.전체인구,
    p.도시인구비율_용도,
    t.월6만이상_PCT,
    ROUND(p.전체인구 * t.월6만이상_PCT / 100, 0) AS 고부담_추정인구수
FROM population p
JOIN (
    SELECT
        지역,
        ROUND(w6to8 + w8to10 + w10to12 + over12, 1) AS 월6만이상_PCT
    FROM transport_cost
    WHERE 지역 != '전체'
) t
ON  p.시도 LIKE '%' || t.지역 || '%'
 OR t.지역 LIKE '%' || SUBSTR(p.시도, 1, 2) || '%'
ORDER BY 고부담_추정인구수 DESC
LIMIT 10;


-- ============================================================
-- 분석 9. 연도별 이용량 요약
-- 목적: 종합 결론 도출을 위한 연도별 핵심 수치 정리
-- ============================================================
SELECT
    STRFTIME('%Y', 사용일자) AS 연도,
    COUNT(DISTINCT 사용일자) AS 데이터일수,
    SUM(승차총승객수 + 하차총승객수) AS 연간_총이용객수,
    ROUND(SUM(승차총승객수 + 하차총승객수) * 1.0
          / COUNT(DISTINCT 사용일자), 0) AS 일평균_이용객수,
    CASE
        WHEN STRFTIME('%Y', 사용일자) IN ('2022','2023') THEN '정책시행전'
        WHEN STRFTIME('%Y', 사용일자) IN ('2024','2025') THEN '정책시행후'
        ELSE '참고'
    END AS 정책구분
FROM subway_usage
WHERE 사용일자 BETWEEN '2022-01-01' AND '2025-12-31'
GROUP BY 연도
ORDER BY 연도;
